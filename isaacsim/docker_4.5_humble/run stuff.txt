BASE="$HOME/docker/isaac-sim"
USD_DIR="/home/wfernando1/Documents/erobs/isaacsim"
ROS_WS="/home/wfernando1/Documents/erobs/isaacsim/docker_4.5_humble/ros_ws"
USD_FILE="erobs-isaacsim.usd"

xhost +local:docker
docker run --rm -it \
  --runtime=nvidia \
  --gpus all \
  --device /dev/dri:/dev/dri \
  --group-add 44 \
  --group-add 110 \
  --device-cgroup-rule='c 226:* rmw' \
  --network host \
  --cap-add SYS_ADMIN \
  --security-opt seccomp=unconfined \
  --security-opt apparmor=unconfined \
  --tmpfs /dev/shm:rw,nosuid,nodev,exec,size=2g \
  -e DISPLAY="$DISPLAY" \
  -e QT_X11_NO_MITSHM=1 \
  -e XDG_RUNTIME_DIR=/tmp/runtime-developer \
  -e NVIDIA_VISIBLE_DEVICES=all \
  -e NVIDIA_DRIVER_CAPABILITIES=all \
  -e __NV_PRIME_RENDER_OFFLOAD=1 \
  -e __GLX_VENDOR_LIBRARY_NAME=nvidia \
  -e LIBGL_DRI3_DISABLE=1 \
  -e ACCEPT_EULA=Y \
  -e PRIVACY_CONSENT=Y \
  -v /tmp/.X11-unix:/tmp/.X11-unix:rw \
  -v /tmp/runtime-developer:/tmp/runtime-developer:rw \
  -v "$BASE/cache/kit":/isaac-sim/kit/cache:rw \
  -v "$BASE/cache/ov":/home/developer/.cache/ov:rw \
  -v "$BASE/cache/warp":/home/developer/.cache/warp:rw \
  -v "$BASE/cache/pip":/home/developer/.cache/pip:rw \
  -v "$BASE/data/ov":/home/developer/.local/share/ov:rw \
  -v "$USD_DIR":/workspace/erobs:rw \
  -v "$ROS_WS":/workspace/ros_ws:rw \
  -v "$USD_DIR/scripts":/workspace/scripts:rw \
  -v "$USD_DIR/ur3e.urdf":/workspace/ros_ws/install/ur3e.urdf:ro \
  isaac-sim:4.5-humble_ros_ws_fullybuilt \
  /isaac-sim/isaac-sim.sh \
    --enable isaacsim.ros2.bridge \
    /workspace/erobs/"$USD_FILE" \
    --/rtx/post/dlss/enabled=false
